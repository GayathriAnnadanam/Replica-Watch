CREATE TABLE nodes (
    node_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    node_name VARCHAR(100) NOT NULL,
    role VARCHAR(20) NOT NULL,   -- primary or standby
    status VARCHAR(20) NOT NULL  -- active, inactive, failed
);

CREATE TABLE replication_logs (
    log_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    node_id BIGINT REFERENCES nodes(node_id),
    lag_ms INT,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE failover_events (
    event_id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    node_id BIGINT REFERENCES nodes(node_id),
    event_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    details TEXT
);

    CASE WHEN x % 5 = 0 THEN 'primary' ELSE 'standby' END,
    CASE WHEN x % 7 = 0 THEN 'inactive' ELSE 'active' END
